# Copies a target's file, and any other files given, into a folder in release on every build. Targets are built in the build directory and copied rather than written straight to release, so that release always holds what was built last even when several build directories, such as a debug and a release one, share it.
function(nvgt_publish target destination)
	set(commands COMMAND "${CMAKE_COMMAND}" -E make_directory "${destination}")
	foreach(file IN ITEMS "$<TARGET_FILE:${target}>" ${ARGN})
		list(APPEND commands COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${file}" "${destination}")
	endforeach()
	file(RELATIVE_PATH shown "${CMAKE_SOURCE_DIR}" "${destination}")
	add_custom_target(${target}_publish ALL ${commands} COMMENT "Copying ${target} to ${shown}" VERBATIM)
	add_dependencies(${target}_publish ${target})
endfunction()

# Links a target that is NVGT or one of its stubs with everything the engine needs. The libraries are linked directly rather than through nvgt_engine so that they come first on the link line, in this order, which matters to GNU linkers.
function(nvgt_link_engine target)
	target_sources(${target} PRIVATE $<TARGET_OBJECTS:nvgt_extra_objects>)
	target_link_libraries(${target} PRIVATE nvgt_core_options ${NVGT_ENGINE_LIBRARIES} nvgt_engine)
	set_target_properties(${target} PROPERTIES LINKER_LANGUAGE CXX PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/pdb")
	if(NVGT_TARGET STREQUAL "macos")
		set_target_properties(${target} PROPERTIES INSTALL_RPATH "@loader_path;@loader_path/lib;@loader_path/../Frameworks")
	elseif(NVGT_TARGET STREQUAL "linux")
		set_target_properties(${target} PROPERTIES INSTALL_RPATH "$ORIGIN/.;$ORIGIN/lib" LINK_DEPENDS "${CMAKE_SOURCE_DIR}/build/linux_exports.txt")
		target_link_options(${target} PRIVATE "-Wl,--dynamic-list=${CMAKE_SOURCE_DIR}/build/linux_exports.txt") # The symbols plugins resolve from the executable, see build/linux_exports.txt.
	endif()
endfunction()

# Links plist and archive, which NVGT uses when compiling games but stubs don't need. They must come after the engine's static libraries, or symbols such as zlib's that these shared libraries also export would be taken from them.
function(nvgt_link_bundling_libraries target)
	if(NVGT_TARGET MATCHES "^(windows|macos|linux)$")
		nvgt_dep_libraries(libraries plist-2.0 archive)
		target_link_libraries(${target} PRIVATE ${libraries})
	endif()
	if(NVGT_TARGET STREQUAL "windows")
		target_link_options(${target} PRIVATE /delayload:plist-2.0.dll /delayload:archive.dll)
	endif()
endfunction()

		endforeach()

		# On Windows we also need a build of NVGT with no console, that is using the windows subsystem. Only nvgt.cpp is compiled again for it.
		add_executable(nvgtw WIN32 src/nvgt.cpp $<TARGET_OBJECTS:nvgt_objects>)
		target_compile_definitions(nvgtw PRIVATE NO_OBFUSCATE NVGT_WIN_APP)

		set_target_properties(nvgtw PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

	endif()

	else()
		set(stub_platform ${NVGT_TARGET})
	endif()
	if(NVGT_UPX AND NVGT_TARGET STREQUAL "windows")
		find_program(NVGT_UPX_EXECUTABLE upx)
	endif()
	foreach(variant IN ITEMS "" _nc)
		set(stub nvgt_stub${variant})
		add_executable(${stub} $<TARGET_OBJECTS:nvgt_stub_objects>)
		nvgt_link_engine(${stub})
		set_target_properties(${stub} PROPERTIES OUTPUT_NAME nvgt_${stub_platform}${variant} SUFFIX .bin RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/stub")
		if(variant)
			set_target_properties(${stub} PROPERTIES NVGT_NO_COMPILER ON)
		endif()
		if(NVGT_TARGET STREQUAL "windows")
			set_target_properties(${stub} PROPERTIES WIN32_EXECUTABLE ON)
		endif()
		set(stub_files "$<TARGET_FILE:${stub}>")
		set(upx_file "")
		if(NVGT_UPX_EXECUTABLE)
			set(upx_file "${CMAKE_BINARY_DIR}/bin/stub/nvgt_${stub_platform}${variant}_upx.bin")
			add_custom_command(TARGET ${stub} POST_BUILD COMMAND "${NVGT_UPX_EXECUTABLE}" --best --lzma -q -f -o "${upx_file}" "$<TARGET_FILE:${stub}>" VERBATIM)
			list(APPEND stub_files "${upx_file}")
		endif()
		foreach(stub_file IN LISTS stub_files)
			add_custom_command(TARGET ${stub} POST_BUILD COMMAND "${CMAKE_COMMAND}" "-DFILE=${stub_file}" -P "${CMAKE_SOURCE_DIR}/cmake/NvgtStampStub.cmake" VERBATIM)
		endforeach()
		nvgt_publish(${stub} "${NVGT_RELEASE_DIR}/stub" ${upx_file})
	endforeach()
endif()
